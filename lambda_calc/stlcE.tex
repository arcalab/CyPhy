\begin{frame}{Recalling Lambda-calculus}
  \begin{flalign*}
      \infer[]{\Gamma \vljud x : \typeA}{x :
        \typeA \in \Gamma} \hspace{3cm}
      \infer[]{\Gamma \vljud \ast : 1}{}  \hspace{3cm}
      \infer[]{\Gamma \vljud \pi_1 \, v : \typeA }
      {\Gamma \vljud v: \typeA \times \typeB}
  \end{flalign*}
  \begin{flalign*}
      \infer[]{\Gamma \vljud \langle v,u \rangle :
        \typeA \times \typeB}
      {\Gamma \vljud v: \typeA \qquad \Gamma \vljud u: \typeB} \hspace{3cm}
     \infer[]{\Gamma \vljud \lambda x : \typeA . \, v : \typeA
        \to \typeB}
      {\Gamma, x : \typeA \vljud v : \typeB}
    \end{flalign*}
   \begin{flalign*}
      \infer[]{\Gamma \vljud v \, u : \typeB}
      {\Gamma \vljud  v : \typeA \to \typeB \quad
        \Gamma \vljud  u : \typeA}
    \end{flalign*}
\end{frame}
%%

\begin{frame}{Sequential Composition Revisited}
  A ``native'' deductive rule
  \begin{flalign*}
    \infer{\Gamma \vljud x \leftarrow v ; u : \typeB}{
      \Gamma \vljud v : \typeA \qquad x : \typeA \vljud u : \typeB}
  \end{flalign*}
  It reads \emph{``bind the computation $v$ to $x$ and then run $u$''}

  \vfill
  Interpretation defined as
  \begin{flalign*}
    \infer{\sem{\Gamma \vljud x \leftarrow v; u : \typeB}\ = g \cdot f}{
      \sem{\Gamma \vljud v : \typeA}\ = f \qquad
      \sem{x : \typeA \vljud u : \typeB}\ = g}
  \end{flalign*}
\end{frame}


\section{Algebraic Operations in Lambda-calculus}

\begin{frame}{Signatures}
  \begin{block}{Signature}
        A set  $\Sigma = \{ (\sigma_1,n_1), (\sigma_2,n_2), \dots \}$ of
        operations $\sigma_i$ paired with the \alert{\textbf{number}} of
        inputs $n_i$ they are supposed to receive
  \end{block}
  
  These constitute the aforementioned \alert{\textbf{algebraic operations}}

  \vfill
  \begin{block}{Examples}
    \begin{itemize}
    \item Exceptions : $\{ (\mathrm{e},0) \}$
    \item Read a bit from the environment : $\{ (\mathrm{read},2) \}$
    \item \alert{\textbf{Wait calls}} : 
            $\{ (\mathrm{wait}_n, 1) \mid n \in \mathbb{N} \}$
    \item Non-deterministic choice : $\{ (\mathrm{+},2) \}$
    \item \alert{\textbf{Trajectories}} : \dots
    \end{itemize}
  \end{block}
\end{frame}
%
%
\begin{frame}{Algebraic Operations in Lambda-calculus}
  We take a signature $\Sigma$ of operations
  and introduce a new rule

  \begin{flalign*}    
      \infer[]{\Gamma \vljud \sigma(m_1,\dots,m_n) : \typeA}
      {(\sigma,n) \in \Sigma \qquad \forall 1 \leq i \leq n. \,
      \> \Gamma \vljud m_i : \typeA}
    \end{flalign*}
\end{frame}
%
\begin{frame}{Examples}
   $x : \typeA \vljud \mathrm{wait}_1(x) : \typeA $ -- adds
                \alert{\textbf{delay}} of one second to returning $x$
                \\[10pt]
   $\Gamma \vljud \mathrm{e}() : \typeA$ -- raises an 
                \alert{\textbf{exception}} $e$
                \\[10pt]
   $\Gamma \vljud \mathrm{write}_v(m) : \typeA$ -- writes 
                $v$ in \alert{\textbf{memory}} and then runs $m$
                \\[10pt]
   $x : \typeA \times \typeA \vljud \mathrm{read}(\pi_1 \> x,
                  \pi_2 \> x) : \typeA$ -- \alert{\textbf{receives}} a bit:
                  if $0$ returns $\pi_1 \> x$ ($\pi_2 \> x$ otherwise)
\end{frame}
%
\begin{frame}{Exercise}

    Define a $\lambda$-term with variable $x$ that requests a bit from the user
    and depending on the value read it returns $x$ with either one or two
    seconds of delay.

\end{frame}

\section{Semantics}
%
\begin{frame}{The Million-Dollar Question}

  How to provide semantics to such programming languages ?

  The short answer: via \alert{\textbf{monads !!}}

  The long answer: see the next slides :-) \dots

\end{frame}
%
\begin{frame}{The Core Idea}
  Programs $\Gamma \vljud v  : \typeA$ interpreted as functions
  \begin{flalign*}
    \sem{\Gamma \vljud v : \typeA}\ : \sem{\Gamma}\ \longrightarrow \sem{\typeA}
  \end{flalign*}

  \dots\ and there exists \alert{\textbf{only one}} function of type
  \begin{flalign*}
   \sem{\Gamma}\ \longrightarrow \sem{1}
  \end{flalign*}

  \alert{\textbf{Problem}}: it is then necessarily the case that
  \begin{flalign*}
    \sem{\Gamma \vljud x : \typeOne}\ = \sem{\Gamma \vljud
      \mathrm{wait_1}(x) : \typeOne}
  \end{flalign*}
  despite these programs having different execution times
\end{frame}
%

\begin{frame}{The Core Idea}
 Interpreted $\Gamma \vljud v : \typeA$ as a function
 \begin{flalign*}
   \sem{\Gamma \vljud v : \typeA}\ : \sem{\Gamma}\ \longrightarrow \sem{\typeA}
 \end{flalign*}
 but values now come with \alert{\textbf{effects}} \dots

 Instead of having $\sem{\typeA}$ as set of outputs, we will have
 set $T\sem{\typeA}$ of \alert{\textbf{effectful values}} 
 \begin{flalign*}
   \sem{\Gamma \vljud m : \typeA}\ : \sem{\Gamma}\ \longrightarrow T\sem{\typeA}
 \end{flalign*}

 $T$ should thus be a \alert{\textbf{set-constructor}}: given a set of
 outputs $X$ it returns a set of effectful values $T X$ over $X$
\end{frame}
%
\begin{frame}{Examples}
  For wait calls, the corresponding set-constructor $T$ is defined as
  \begin{flalign*}
    X \mapsto \mathbb{N} \times X
  \end{flalign*}
  i.e. values in $X$ paired with an \alert{\textbf{execution time}}
  \vfill
  For exceptions, the corresponding set-constructor $T$ is defined as
  \begin{flalign*}
    X \mapsto X + \{ e \}
  \end{flalign*}
  i.e. values in $X$ plus an element $e$ \alert{\textbf{representing 
  the exception}}
\end{frame}

%
\begin{frame}{Another Problem}
  This idea of a set-constructor $T$ looks good, but  \dots\

  \dots\ it breaks sequential composition
  \begin{flalign*}
    \sem{\Gamma \vljud m : \typeA}\ &:\ \sem{\Gamma}\ \, \longrightarrow
    \alert{T\mathbf{\sem{\typeA}}} \\
    \sem{x : \typeA \vljud n : \typeB}\ & :\ \alert{\mathbf{\sem{\typeA}}}\ 
    \longrightarrow T\sem{\typeB}
  \end{flalign*}
    
  \medskip
  \pause
  We will need to \alert{\textbf{lift}} 
  a function $h : X \to TY$ to one of type
  \begin{flalign*}
    h^\star : TX \to TY
  \end{flalign*}
\end{frame}
%
\begin{frame}{Examples}

  There are set-constructors $T$ for which this is possible

  \vspace{0.5cm}
  In the case of \alert{\textbf{wait-calls}}
  \begin{flalign*}
    \infer{f^\star(n,x) = (n+m,y) \text{ where } f(x) = (m,y)}
    {f : X \to TY = \mathbb{N} \times Y}
  \end{flalign*}

  \vspace{0.5cm}
  In the case of \alert{\textbf{exceptions}}
  \begin{flalign*}
      \infer{f^\star(x) = f (y) \qquad f^\star(e) = e}{f : X \to TY = Y + \{e\}}
  \end{flalign*}
\end{frame}
%
\begin{frame}{Testing the Idea\ \dots}
  \begin{flalign*}
   & \, \sem{x : \typeOne \vljud y \leftarrow \mathrm{wait_1}(x) ;
   \mathrm{wait_2}(y) : \typeOne} \\[0.2cm]
   & =\  \sem{ y : \typeOne \vljud \mathrm{wait_2}(y) : \typeOne }^\ast\ \cdot
   \sem{x : \typeOne \vljud \mathrm{wait_1}(x) : \typeOne} \\[0.2cm]
   & =\ (v \mapsto (2,v))^\ast\ \cdot (v \mapsto (1,v)) \\[0.2cm]
   & =\ v \mapsto (3,v)
  \end{flalign*}
\end{frame}


\begin{frame}{Yet Another Problem}

  Idea of interpreting $\lambda$-terms $\Gamma \vljud m : \typeA$ as functions
  \begin{flalign*}
   \sem{\Gamma \vljud m : \typeA} : \sem{\Gamma}\ \longrightarrow T\sem{\typeA}
  \end{flalign*}
  looks good, but it presupposes that all terms invoke
  effects

  \pause
  \vfill
  Some terms do not do this, e.g.
  \begin{flalign*}
    \sem{x: \typeA \vljud x : \typeA} : \sem{\typeA}\ \longrightarrow \sem{\typeA}
  \end{flalign*}

  \begin{block}{Solution}
   $T\sem{\typeA}$ should include \alert{\textbf{effect-free}} values, 
   and we should have 
   \begin{flalign*}
    \eta_{\sem{\typeA}} : \sem{\typeA}\ \longrightarrow T\sem{\typeA}
   \end{flalign*}
   which maps a value to its effect-free representation
  \end{block}  
\end{frame}


\begin{frame}{Examples}

  In the case of \alert{\textbf{wait-calls}}
  \begin{flalign*}
    \infer{\eta_X (x) = (0,x)}
    {TX = \mathbb{N} \times X}
  \end{flalign*}
  (i.e. no wait call invoked)
  
  \vspace{1cm}
  In the case of \alert{\textbf{exceptions}}
  \begin{flalign*}
      \infer{\eta_X (x) = x}{TX = X + \{e\}}
  \end{flalign*}
  (i.e. no exception $e$ raised) 

\end{frame}
%
\begin{frame}{Monads Unlocked !!}
  Previous analysis leads to the notion of a \alert{\textbf{monad}}

  \pause
  \bigskip
  \begin{block}{Monad}
    Triple $(T,\eta,(-)^\star)$ where $T$ is a
    set-constructor, $\eta$ a function
    $\eta_X : X \to TX$ for each set $X$, and $(-)^\star$
    an operation
    \begin{flalign*}
      \infer{f^\star : TX \to TY}{f : X \to TY}
    \end{flalign*}
    s.t. the following laws hold:
    $\eta^{\star}=\mathrm{id}$, %&&
    $f^{\star}\cdot \eta=f$, %&&
    $(f^{\star} \cdot g)^{\star}=f^{\star} \cdot g^{\star}$
  \end{block}

  {\small These laws are required to forbid ``weird''
  computational behaviour}
\end{frame}

\begin{frame}{To Keep In Mind}
 Recall that
  \begin{itemize}
  \item we fixed a signature $\Sigma$ of algebraic operations
          \\[10pt]
  \item we now have monads at our disposal
          \\[10pt]
  \item programs $\Gamma \vljud v : \typeA$ can be seen either as
    functions of type $\sem{\Gamma}\ \to \sem{\typeA}$ or of 
    type $\sem{\Gamma}\ \to T\sem{\typeA}$
  \end{itemize}
\end{frame}

\begin{frame}{Semantics}
 Types $\typeA$ interpreted as sets $\sem{\typeA}$
 \begin{flalign*}
    \sem{1}\ = \{ \star \} \hspace{.7cm}
    \sem{\typeA \times \typeB}\ = \sem{\typeA} \times \sem{\typeB} \hspace{0.7cm}
   \sem{\typeA \to \typeB}\ = (T\sem{\typeB})^{\sem{\typeA}} 
 \end{flalign*}
 \vfill
 Typing contexts $\Gamma$ interpreted as
 \begin{flalign*}
  \sem{\Gamma}\ = \sem{x_1 : \typeA_1, \dots ,x_n : \typeA_n}\ =
 \sem{\typeA_1} \times \dots \times \sem{\typeA_n} 
 \end{flalign*}
 \vfill
 For each $(\sigma,n) \in \Sigma$ and set $X$ we postulate
 the existence of a map
 \begin{flalign*}
   \sem{\sigma}_X \, : (TX)^n \longrightarrow TX
 \end{flalign*}
\end{frame}

\begin{frame}{Semantics}
        \footnotesize{
   \begin{flalign*}
    \infer{\sem{\Gamma \vljud x_i}\ = \pi_i}{x_i : \typeA \in \Gamma}
    \hspace{1.5cm}
    \infer{\sem{\Gamma \vljud \ast}\ = \> !}{}
    \hspace{1.5cm}
    \infer{\sem{\Gamma \vljud \langle v, u \rangle : \typeA \times \typeB}\ 
    = \langle f,g \rangle}
    {\sem{\Gamma \vljud v : \typeA} = f 
    \qquad \sem{\Gamma \vljud u : \typeB}\ = g}
  \end{flalign*}
  %%
  \begin{flalign*}
       \infer[]{\sem{\Gamma \vljud \lambda x : \typeA . \, m : \typeA
        \to \typeB}\ = \lambda f}
    {\sem{\Gamma, x : \typeA \cojud m : \typeB}\ = f} \hspace{2.5cm}
          \infer[]{\sem{\Gamma \vljud \pi_1 \, v : \typeA}\ = \pi_1 \cdot f }
      {\sem{\Gamma \vljud v: \typeA \times \typeB}\ = f}
  \end{flalign*}
  \dotfill
  \begin{flalign*}
    \infer{\sem{\Gamma \cojud \ret v : \typeA}\ = \eta \cdot f}
    {\sem{\Gamma \vljud v : \typeA}\ = f} \hspace{2cm}
    \infer{\sem{\Gamma \cojud x \leftarrow m \pcomp n : \typeB} = g^\star \cdot f}
    {\sem{\Gamma \cojud m : \typeA} = f \qquad \sem{x:\typeA \cojud n : \typeB} = g}
  \end{flalign*}
  %%
  \begin{flalign*}
     \infer[]{\sem{\Gamma \cojud v \, u : \typeB}\ = \mathrm{app} \cdot 
     \langle f, g \rangle}
     {\sem{\Gamma \vljud  v : \typeA \to \typeB}\ = f \quad
     \sem{\Gamma \vljud  u : \typeA}\ = g} \\[2.5ex]
     \infer{\sem{\Gamma \cojud \sigma(m_1, \dots m_n) : \typeA}\ = 
     \sem{\sigma}_{\sem{\typeA}}
     \, \cdot \, \langle f_1,\dots,f_n \rangle}{(\sigma, n) \in \Sigma \qquad \forall 1 \leq i
     \leq n . \> \sem{\Gamma \cojud m_i : \typeA} = f_i}
  \end{flalign*}
}
\end{frame}
